18template <class FUNCTION, class... BOUND_ARGS>
19struct terminating_partial {
20 FUNCTION function;
21 std::tuple<BOUND_ARGS...> bound_args;
22
23 template <class NEXT_ARG>
24 auto operator()(NEXT_ARG&& next_arg) {
25 return invoke_or_extend(std::forward<NEXT_ARG>(next_arg), std::index_sequence_for<BOUND_ARGS...>{});
26 }
27
28 template <class NEXT_ARG>
29 auto operator()(NEXT_ARG&& next_arg) const {
30 return invoke_or_extend_const(std::forward<NEXT_ARG>(next_arg), std::index_sequence_for<BOUND_ARGS...>{});
31 }
32
33 private:
34 template <class NEXT_ARG, std::size_t... IDX>
35 auto invoke_or_extend(NEXT_ARG&& next_arg, std::index_sequence<IDX...>) {
36 if constexpr (std::invocable<FUNCTION&, BOUND_ARGS&..., NEXT_ARG>) {
37 return std::invoke(function, std::get<IDX>(bound_args)..., std::forward<NEXT_ARG>(next_arg));
38 } else {
39 using NEXT_PARTIAL = terminating_partial<FUNCTION, BOUND_ARGS..., remove_cvref_t<NEXT_ARG>>;
40 return NEXT_PARTIAL{
41 function,
42 std::tuple_cat(std::move(bound_args),
43 std::tuple<remove_cvref_t<NEXT_ARG>>{std::forward<NEXT_ARG>(next_arg)})};
44 }
45 }
46
47 template <class NEXT_ARG, std::size_t... IDX>
48 auto invoke_or_extend_const(NEXT_ARG&& next_arg, std::index_sequence<IDX...>) const {
49 if constexpr (std::invocable<const FUNCTION&, const BOUND_ARGS&..., NEXT_ARG>) {
50 return std::invoke(function, std::get<IDX>(bound_args)..., std::forward<NEXT_ARG>(next_arg));
51 } else {
52 using NEXT_PARTIAL = terminating_partial<FUNCTION, BOUND_ARGS..., remove_cvref_t<NEXT_ARG>>;
53 return NEXT_PARTIAL{
54 function,
55 std::tuple_cat(bound_args, std::tuple<remove_cvref_t<NEXT_ARG>>{std::forward<NEXT_ARG>(next_arg)})};
56 }
57 }
58};

68// Applicative pattern invariants:
69// - Minimal required operations are pure and apply.
70// - invoke is derived from pure/apply via terminating partial application,
71// but an Impl may provide a custom invoke for different semantics or
72// efficiency.
73// - Derived operations (lift/ap/zip_with/discard_*) live on that object.
74// - Dispatch happens through a provided object or
75// applicative_typeclass<Concrete>.
76// - Do not introduce hidden alternate semantics without a distinct map/type.
77

113 template <class FUNCTION, class FIRST_ARGUMENT, class... REST_ARGUMENTS>
114 auto invoke(this auto&& self,
115 FUNCTION&& function,
116 FIRST_ARGUMENT&& first_argument,
117 REST_ARGUMENTS&&... rest_arguments) {
118 using SELF = std::remove_reference_t<decltype(self)>;
119 using IMPL_BASE = std::conditional_t<std::is_const_v<SELF>, const Impl, Impl>;
120
121 if constexpr (requires(IMPL_BASE& impl) {
122 impl.invoke(std::forward<FUNCTION>(function),
123 std::forward<FIRST_ARGUMENT>(first_argument),
124 std::forward<REST_ARGUMENTS>(rest_arguments)...);
125 }) {
126 return static_cast<IMPL_BASE&>(self).invoke(std::forward<FUNCTION>(function),
127 std::forward<FIRST_ARGUMENT>(first_argument),
128 std::forward<REST_ARGUMENTS>(rest_arguments)...);
129 } else {
130 auto lifted_function = self.pure(detail::make_terminating_partial(std::forward<FUNCTION>(function)));
131 return self.apply_chain(self.ap(std::move(lifted_function), std::forward<FIRST_ARGUMENT>(first_argument)),
132 std::forward<REST_ARGUMENTS>(rest_arguments)...);
133 }
134 }
135 // a11f7d8b-8f89-4f3e-9c92-f9f08ab7ef11 end

143 template <class ACCUMULATED, class NEXT_ARGUMENT, class... REST_ARGUMENTS>
144 auto apply_chain(this auto&& self,
145 ACCUMULATED&& accumulated,
146 NEXT_ARGUMENT&& next_argument,
147 REST_ARGUMENTS&&... rest_arguments) {
148 auto next = self.ap(std::forward<ACCUMULATED>(accumulated), std::forward<NEXT_ARGUMENT>(next_argument));
149 if constexpr (sizeof...(REST_ARGUMENTS) == 0) {
150 return next;
151 } else {
152 return self.apply_chain(std::move(next), std::forward<REST_ARGUMENTS>(rest_arguments)...);
153 }

258template <class VALUE_TYPE>
259struct OptionalApplicativeImpl {
260 template <class VALUE>
261 auto pure(this auto&&, VALUE&& value) -> std::optional<remove_cvref_t<VALUE>> {
262 return std::optional<remove_cvref_t<VALUE>>{std::forward<VALUE>(value)};
263 }
264
265 template <class FUNCTION_IN_CONTEXT, class ARGUMENT_IN_CONTEXT>
266 auto apply(this auto&&, FUNCTION_IN_CONTEXT&& function, ARGUMENT_IN_CONTEXT&& argument) {
267 using Result = std::invoke_result_t<decltype(*function), decltype(*argument)>;
268
269 if (!function || !argument) {
270 return std::optional<remove_cvref_t<Result>>{};
271 }
272
273 return std::optional<remove_cvref_t<Result>>{
274 std::invoke(*std::forward<FUNCTION_IN_CONTEXT>(function), *std::forward<ARGUMENT_IN_CONTEXT>(argument))};
275 }
276};

278template <class VALUE_TYPE>
279struct OptionalApplicativeMap : Applicative<OptionalApplicativeImpl<VALUE_TYPE>> {
280 using OptionalApplicativeImpl<VALUE_TYPE>::apply;
281 using OptionalApplicativeImpl<VALUE_TYPE>::pure;
282};

285template <class VALUE_TYPE>
286inline constexpr auto applicative_typeclass<std::optional<VALUE_TYPE>> = OptionalApplicativeMap<VALUE_TYPE>{};
