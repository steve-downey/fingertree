5// DualMonoid<M> wraps a monoidal value M and flips the argument order of
6// combine. The identity element is the same as M's identity.
7//
8// This is the standard algebraic dual: if (M, ·, e) is a monoid then
9// (M, ·ᵒᵖ, e) where a ·ᵒᵖ b = b · a is also a monoid.
10//
11// Primary use: FingerTree5::reversed() returns a tree whose measure tag is
12// DualMonoid<Tag>, ensuring that prefix-accumulated measures grow from the
13// new left edge (the old right edge) under the correct combine order.
14//
15// split_at_measure requires >= on the Tag type; DualMonoid<M> forwards >=
16// from M so that split operations work on reversed trees.
17
18#include <beman/fingertree/typeclass/monoid.hpp>

22namespace beman::fingertree::typeclass {

24template <typename M>
25struct DualMonoid {
26 M value;
27 friend auto operator==(const DualMonoid&, const DualMonoid&) -> bool = default;
28
29 // Forward >= from M so split_at_measure compiles on reversed trees.
30 friend auto operator>=(const DualMonoid& lhs, const DualMonoid& rhs) -> bool
31 requires requires(const M& a, const M& b) {
32 { a >= b } -> std::convertible_to<bool>;
33 }
34 {
35 return lhs.value >= rhs.value;
36 }
37};
38
39template <typename M>
40struct Monoid<DualMonoid<M>> {
41 constexpr auto identity() const -> DualMonoid<M> { return DualMonoid<M>{monoid_v<M>.identity()}; }
42 constexpr auto combine(const DualMonoid<M>& lhs, const DualMonoid<M>& rhs) const -> DualMonoid<M> {
43 // Flip the argument order — that's the entire mechanism.
44 return DualMonoid<M>{monoid_v<M>.combine(rhs.value, lhs.value)};
45 }
46};
