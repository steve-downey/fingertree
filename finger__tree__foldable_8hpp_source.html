18struct FingerTreeFoldableImpl {

21 // The result monoid is an instance object over the result type, passed
22 // by value and distinct from the tree's own measure monoid MONOID; the
23 // Impl never chooses one (beman.transpose
24 // docs/decisions.md#monoid-selection).
25 template <class F, class RESULT_MONOID>
26 auto fold_map(this auto&&,

28 const beman::fingertree::tree::FingerTree<T, MONOID, MEASURE_POLICY, ALLOCATOR>& tree,
29 RESULT_MONOID monoid) -> remove_cvref_t<std::invoke_result_t<F&, const T&>> {
30 using Result = remove_cvref_t<std::invoke_result_t<F&, const T&>>;
31
32 Result acc = monoid.identity();
33 tree.for_each([&](const T& value) { acc = monoid.combine(std::move(acc), std::invoke(function, value)); });

35 }
