53template <typename PAYLOAD_TYPE,
54 typename Tree = FingerTree<Interval<PAYLOAD_TYPE>, IntervalMonoid, IntervalMeasure<PAYLOAD_TYPE>>>
55class FingerTreeIntervalIndex {
56 using Entry = Interval<PAYLOAD_TYPE>;
57
58 Tree d_tree;
59
60 explicit FingerTreeIntervalIndex(Tree tree) : d_tree(std::move(tree)) {}
61
62 public:
63 FingerTreeIntervalIndex() : d_tree(Tree{}) {}
64
65 static auto from_intervals(std::vector<Entry> entries) -> FingerTreeIntervalIndex {
66 return FingerTreeIntervalIndex{Tree::from_sequence(std::move(entries))};
67 }
68
69 auto insert(Entry entry) const -> FingerTreeIntervalIndex {
70 return FingerTreeIntervalIndex{d_tree.snoc(std::move(entry))};
71 }
72
73 auto query_point(std::size_t point) const -> std::vector<PAYLOAD_TYPE> {
74 std::vector<PAYLOAD_TYPE> out;
75 auto parts = d_tree.split_at([point](std::size_t max_end) { return max_end > point; });
76 parts.d_right.for_each([&](const Entry& e) {
77 if (e.d_start <= point && point < e.d_end)
78 out.push_back(e.d_payload);
79 });
80 return out;
81 }
82
83 auto query_overlap(std::size_t start, std::size_t end) const -> std::vector<PAYLOAD_TYPE> {
84 std::vector<PAYLOAD_TYPE> out;
85 auto parts = d_tree.split_at([start](std::size_t max_end) { return max_end > start; });
86 parts.d_right.for_each([&](const Entry& e) {
87 if (e.d_start < end && start < e.d_end)
88 out.push_back(e.d_payload);
89 });
90 return out;
91 }
92
93 auto entries() const -> std::vector<Entry> { return d_tree.flatten(); }
94};

104 // The monoid is an instance object over the result type, passed by
105 // value; the Impl never chooses one (beman.transpose
106 // docs/decisions.md#monoid-selection).
107 template <class F, class RESULT_MONOID>
108 auto fold_map(this auto&&,
109 F&& function,
110 const beman::fingertree::tree::FingerTreeIntervalIndex<PAYLOAD_TYPE, Tree>& index,
111 RESULT_MONOID monoid) -> remove_cvref_t<std::invoke_result_t<F&, const PAYLOAD_TYPE&>> {
112 using Result = remove_cvref_t<std::invoke_result_t<F&, const PAYLOAD_TYPE&>>;
113 return std::ranges::fold_left(index.entries(), Result(monoid.identity()), [&](Result acc, const auto& value) {
114 return monoid.combine(std::move(acc), std::invoke(function, value.d_payload));
115 });
116 }
117};

130 template <class APPLICATIVE, class F>
131 auto traverse(this auto&&,
132 const APPLICATIVE& applicative,
133 F&& function,
134 const beman::fingertree::tree::FingerTreeIntervalIndex<PAYLOAD_TYPE, Tree>& index) {
135 using Context = remove_cvref_t<std::invoke_result_t<F, const PAYLOAD_TYPE&>>;
136 using U = beman::fingertree::applicative_value_t<Context>;
137
138 auto accumulated = applicative.pure(std::vector<beman::fingertree::tree::Interval<U>>{});
139
140 for (const auto& entry : index.entries()) {
141 auto lifted = std::invoke(function, entry.d_payload);
142 accumulated = applicative.invoke(
143 [start = entry.d_start, end = entry.d_end](std::vector<beman::fingertree::tree::Interval<U>> values,
144 U payload) {
145 values.push_back(beman::fingertree::tree::Interval<U>{start, end, std::move(payload)});
146 return values;
147 },
148 std::move(accumulated),
149 std::move(lifted));
150 }
151
152 return applicative.invoke(
153 [](std::vector<beman::fingertree::tree::Interval<U>> values) {
154 return beman::fingertree::tree::FingerTreeIntervalIndex<U>::from_intervals(std::move(values));
155 },
156 std::move(accumulated));
157 }
158};
