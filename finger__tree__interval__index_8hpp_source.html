51template <typename PAYLOAD_TYPE,
52 typename Tree =

54class FingerTreeIntervalIndex {
55 using Entry = Interval<PAYLOAD_TYPE>;
56
57 Tree d_tree;
58
59 explicit FingerTreeIntervalIndex(Tree tree) : d_tree(std::move(tree)) {}
60
61 public:
62 FingerTreeIntervalIndex() : d_tree(Tree{}) {}
63
64 static auto from_intervals(std::vector<Entry> entries) -> FingerTreeIntervalIndex {
65 return FingerTreeIntervalIndex{Tree::from_sequence(std::move(entries))};
66 }
67
68 auto insert(Entry entry) const -> FingerTreeIntervalIndex {
69 return FingerTreeIntervalIndex{d_tree.snoc(std::move(entry))};
70 }
71
72 auto query_point(std::size_t point) const -> std::vector<PAYLOAD_TYPE> {
73 std::vector<PAYLOAD_TYPE> out;
74 auto parts = d_tree.split_at(
75 [point](const IntervalMaxEndTag<PAYLOAD_TYPE>& prefix) { return prefix.d_max_end > point; });
76 parts.d_right.for_each([&](const Entry& e) {
77 if (e.d_start <= point && point < e.d_end)
78 out.push_back(e.d_payload);
79 });
80 return out;
81 }
82
83 auto query_overlap(std::size_t start, std::size_t end) const -> std::vector<PAYLOAD_TYPE> {
84 std::vector<PAYLOAD_TYPE> out;
85 auto parts = d_tree.split_at(
86 [start](const IntervalMaxEndTag<PAYLOAD_TYPE>& prefix) { return prefix.d_max_end > start; });
87 parts.d_right.for_each([&](const Entry& e) {
88 if (e.d_start < end && start < e.d_end)
89 out.push_back(e.d_payload);
90 });
91 return out;
92 }
93
94 auto entries() const -> std::vector<Entry> { return d_tree.flatten(); }

162 },
163 std::move(accumulated),
