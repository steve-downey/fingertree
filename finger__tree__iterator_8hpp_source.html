5// Bidirectional iterator for FingerTree.
6//
7// Include this header to obtain iteration support:
8//
9// #include <beman/fingertree/tree/finger_tree_iterator.hpp>
10// for (auto x : tree) { ... } // range-based for via ADL
11// auto it = beman::fingertree::tree::begin(tree); // explicit free function
12//
13// Free functions begin(t)/end(t) work for any <T,Monoid,Measure> combination.
14// For the default measure <T, std::size_t, UnitMeasure<T,std::size_t>>,
15// tree size is computed in O(1) via measure(). For custom measures the size
16// is computed in O(N) by counting elements through for_each.
17//
18// Iterator category: bidirectional (operator++ and operator-- are O(1)
19// amortized). Random-access is deliberately not claimed — tagging RA would
20// lie to std::sort about the cost of distance and advance.
21//
22// Iterator validity: the iterator holds a shared_ptr to the root, so the
23// tree can be destroyed while iterators remain valid. Structural sharing
24// means all nodes reachable from the iterator's snapshot stay alive.
25//
26// Comparing iterators from different trees is undefined behaviour.
27

40template <typename T, typename MONOID, typename MP, typename Alloc>
41class FingerTreeIterator {
42 using FT = FingerTree<T, MONOID, MP, Alloc>;
43 using EP = ft::ElemPtr<T, MONOID>;
44 using E = ft::Elem<T, MONOID>;
45
46 // -----------------------------------------------------------------------
47 // State
48 // -----------------------------------------------------------------------
49
50 struct SpineFrame {
51 std::shared_ptr<const typename FT::Deep> deep;
52 enum class Section : std::uint8_t { LEFT = 0, SPINE = 1, RIGHT = 2 } section;
53 std::size_t index; // digit index (unused when section == SPINE)
54 };
55
56 // Shorthand — dependent type so using enum is not allowed, use these.
57 static constexpr auto k_left = SpineFrame::Section::LEFT;
58 static constexpr auto k_spine = SpineFrame::Section::SPINE;
59 static constexpr auto k_right = SpineFrame::Section::RIGHT;
60
61 struct ElemFrame {
62 EP elem;
63 std::size_t child; // 0/1 for Node2, 0/1/2 for Node3
64 };
65
66 std::vector<SpineFrame> d_spine_path; // outermost → innermost Deep
67 std::vector<ElemFrame> d_elem_path; // outer → leaf-parent Node
68 EP d_current_leaf; // null at end()
69 std::size_t d_absolute_index{0};
70 std::size_t d_tree_size{0};
71 std::shared_ptr<const FT> d_root_keepalive;
72
73 // -----------------------------------------------------------------------
74 // Element-level helpers
75 // -----------------------------------------------------------------------
76
77 static auto num_children(const EP& ep) -> std::size_t {
78 return std::visit(ft::overloaded{[](const typename E::Leaf&) -> std::size_t { return 0; },
79 [](const typename E::Node2&) -> std::size_t { return 2; },
80 [](const typename E::Node3&) -> std::size_t { return 3; }},
81 ep->d_data);
82 }
83
84 static auto nth_child(const EP& ep, std::size_t n) -> EP {
85 return std::visit(ft::overloaded{[](const typename E::Leaf&) -> EP {
86 assert(false && "nth_child called on Leaf");
87 std::unreachable();
88 },
89 [n](const typename E::Node2& nd) -> EP { return n == 0 ? nd.a : nd.b; },
90 [n](const typename E::Node3& nd) -> EP {
91 return n == 0 ? nd.a : n == 1 ? nd.b : nd.c;
92 }},
93 ep->d_data);
94 }
95
96 // Walk down to the leftmost Leaf within ep, pushing ElemFrames.
97 void descend_elem_left(EP ep) {
98 while (!ft::is_leaf(ep)) {
99 d_elem_path.push_back({ep, 0});
100 ep = nth_child(ep, 0);
101 }
102 d_current_leaf = std::move(ep);
103 }
104
105 // Walk down to the rightmost Leaf within ep, pushing ElemFrames.
106 void descend_elem_right(EP ep) {
107 while (!ft::is_leaf(ep)) {
108 std::size_t last = num_children(ep) - 1;
109 d_elem_path.push_back({ep, last});
110 ep = nth_child(ep, last);
111 }
112 d_current_leaf = std::move(ep);
113 }
114
115 // -----------------------------------------------------------------------
116 // Spine-level descent helpers
117 // -----------------------------------------------------------------------
118
119 // Called when the top SpineFrame's LEFT digit is exhausted (going forward).
120 // Sets the top frame's section to SPINE and descends into the spine.
121 void enter_spine_left(const typename FT::SpinePtr& sp) {
122 // Mark that we are now in the spine section of the current Deep.
123 d_spine_path.back().section = k_spine;
124
125 if (!sp || sp->is_empty()) {
126 // No spine — go directly to RIGHT.
127 d_spine_path.back().section = k_right;
128 d_spine_path.back().index = 0;
129 descend_elem_left(d_spine_path.back().deep->d_right[0]);
130 return;
131 }
132 std::visit(ft::overloaded{[](const typename FT::Empty&) {},
133 [this](const typename FT::Single& s) {
134 // No SpineFrame pushed; descend via
135 // ElemFrames only.
136 descend_elem_left(s.d_elem);
137 },
138 [this](const typename FT::DeepPtr& d) {
139 // reserve(32) in constructor prevents
140 // reallocation.
141 d_spine_path.push_back({d, k_left, 0});
142 descend_elem_left(d_spine_path.back().deep->d_left[0]);
143 }},
144 sp->d_repr);
145 }
146
147 // Called when the top SpineFrame's RIGHT digit is exhausted (going
148 // backward).
149 void enter_spine_right(const typename FT::SpinePtr& sp) {
150 d_spine_path.back().section = k_spine;
151
152 if (!sp || sp->is_empty()) {
153 // No spine — go directly to LEFT.
154 auto sz = d_spine_path.back().deep->d_left.size();
155 d_spine_path.back().section = k_left;
156 d_spine_path.back().index = sz - 1;
157 descend_elem_right(d_spine_path.back().deep->d_left.back());
158 return;
159 }
160 std::visit(ft::overloaded{[](const typename FT::Empty&) {},
161 [this](const typename FT::Single& s) { descend_elem_right(s.d_elem); },
162 [this](const typename FT::DeepPtr& d) {
163 d_spine_path.push_back({d, k_right, d->d_right.size() - 1});
164 descend_elem_right(d_spine_path.back().deep->d_right.back());
165 }},
166 sp->d_repr);
167 }
168
169 // -----------------------------------------------------------------------
170 // Spine-level advance (operator++)
171 // -----------------------------------------------------------------------
172
173 void advance_spine() {
174 while (!d_spine_path.empty()) {
175 auto& top = d_spine_path.back();
176
177 if (top.section == k_left) {
178 if (top.index + 1 < top.deep->d_left.size()) {
179 ++top.index;
180 descend_elem_left(top.deep->d_left[top.index]);
181 return;
182 }
183 // Left digit exhausted — enter the spine.
184 enter_spine_left(top.deep->d_spine);
185 return;
186 }
187
188 if (top.section == k_spine) {
189 // Returned from the spine subtree — move to RIGHT.
190 top.section = k_right;
191 top.index = 0;
192 descend_elem_left(top.deep->d_right[0]);
193 return;
194 }
195
196 // RIGHT section.
197 if (top.index + 1 < top.deep->d_right.size()) {
198 ++top.index;
199 descend_elem_left(top.deep->d_right[top.index]);
200 return;
201 }
202 // Right digit exhausted — pop back to parent's SPINE section.
203 d_spine_path.pop_back();
204 }
205 // All frames exhausted: we are now at end().
206 d_current_leaf = nullptr;
207 d_absolute_index = d_tree_size;
208 }
209
210 // -----------------------------------------------------------------------
211 // Spine-level retreat (operator--)
212 // -----------------------------------------------------------------------
213
214 void retreat_spine() {
215 while (!d_spine_path.empty()) {
216 auto& top = d_spine_path.back();
217
218 if (top.section == k_right) {
219 if (top.index > 0) {
220 --top.index;
221 descend_elem_right(top.deep->d_right[top.index]);
222 return;
223 }
224 // Right beginning — enter spine from right.
225 enter_spine_right(top.deep->d_spine);
226 return;
227 }
228
229 if (top.section == k_spine) {
230 // Returned from the spine subtree — move to LEFT (last
231 // element).
232 auto sz = top.deep->d_left.size();
233 top.section = k_left;
234 top.index = sz - 1;
235 descend_elem_right(top.deep->d_left.back());
236 return;
237 }
238
239 // LEFT section.
240 if (top.index > 0) {
241 --top.index;
242 descend_elem_right(top.deep->d_left[top.index]);
243 return;
244 }
245 // Left beginning — pop back to parent's SPINE section.
246 d_spine_path.pop_back();
247 }
248 // Decremented past begin — undefined behaviour; assert in debug.
249 assert(false && "FingerTreeIterator: decremented begin iterator");
250 std::unreachable();
251 }
252
253 // -----------------------------------------------------------------------
254 // Descent from the rightmost leaf of the full tree (end() -- )
255 // -----------------------------------------------------------------------
256
257 void descend_right_from_root() {
258 std::visit(ft::overloaded{[](const typename FT::Empty&) {
259 assert(false && "decremented end() of empty tree");
260 std::unreachable();
261 },
262 [this](const typename FT::Single& s) { descend_elem_right(s.d_elem); },
263 [this](const typename FT::DeepPtr& d) {
264 d_spine_path.push_back({d, k_right, d->d_right.size() - 1});
265 descend_elem_right(d_spine_path.back().deep->d_right.back());
266 }},
267 d_root_keepalive->d_repr);
268 }
269
270 public:
271 // -----------------------------------------------------------------------
272 // Iterator concept requirements
273 // -----------------------------------------------------------------------
274
275 // Default-constructed iterator is in a singular (unspecified) state.
276 // Required by std::default_initializable (via std::weakly_incrementable).
277 FingerTreeIterator() = default;

289 static auto make_begin(const FT& tree, std::size_t n) -> FingerTreeIterator {
290 FingerTreeIterator it;
291 it.d_tree_size = n;
292 it.d_root_keepalive = std::make_shared<const FT>(tree);
293 it.d_absolute_index = 0;
294 it.d_spine_path.reserve(32); // spine depth ≤ floor(log2(N)) ≤ 30 for N<2^31
295 it.d_elem_path.reserve(32);
296
297 std::visit(ft::overloaded{[&](const typename FT::Empty&) {
298 it.d_current_leaf = nullptr;
299 it.d_absolute_index = n; // consistent with end()
300 },
301 [&](const typename FT::Single& s) { it.descend_elem_left(s.d_elem); },
302 [&](const typename FT::DeepPtr& d) {
303 it.d_spine_path.push_back({d, k_left, 0});
304 it.descend_elem_left(d->d_left[0]);
305 }},
306 tree.d_repr);
307
308 return it;
309 }
310
311 static auto make_end(const FT& tree, std::size_t n) -> FingerTreeIterator {
312 FingerTreeIterator it;
313 it.d_tree_size = n;
314 it.d_root_keepalive = std::make_shared<const FT>(tree);
315 it.d_current_leaf = nullptr;
316 it.d_absolute_index = n;
317 it.d_spine_path.reserve(32);
318 it.d_elem_path.reserve(32);
319 return it;
320 }

322 // -----------------------------------------------------------------------
323 // Dereference
324 // -----------------------------------------------------------------------
325
326 auto operator*() const -> reference {
327 assert(d_current_leaf && "dereferenced end() iterator");
328 return ft::leaf_value(d_current_leaf);
329 }
330
331 auto operator->() const -> pointer {
332 assert(d_current_leaf && "dereferenced end() iterator");
333 return &ft::leaf_value(d_current_leaf);
334 }
335
336 // -----------------------------------------------------------------------
337 // Advance
338 // -----------------------------------------------------------------------
339
340 auto operator++() -> FingerTreeIterator& {
341 assert(d_current_leaf && "incremented end() iterator");
342 ++d_absolute_index;
343
344 // Step 1: advance within the current Node tree.
345 while (!d_elem_path.empty()) {
346 auto& top = d_elem_path.back();
347 if (top.child + 1 < num_children(top.elem)) {
348 ++top.child;
349 descend_elem_left(nth_child(top.elem, top.child));
350 return *this;
351 }
352 d_elem_path.pop_back();
353 }
354 // Step 2: ElemFrames exhausted — advance across the spine structure.
355 advance_spine();
356 return *this;
357 }
358
359 auto operator++(int) -> FingerTreeIterator {
360 auto copy = *this;
361 ++*this;
362 return copy;
363 }
364
365 // -----------------------------------------------------------------------
366 // Retreat
367 // -----------------------------------------------------------------------
368
369 auto operator--() -> FingerTreeIterator& {
370 if (!d_current_leaf) {
371 // end() -- : descend to the rightmost leaf.
372 descend_right_from_root();
373 d_absolute_index = d_tree_size - 1;
374 return *this;
375 }
376 --d_absolute_index;
377
378 // Step 1: retreat within the current Node tree.
379 while (!d_elem_path.empty()) {
380 auto& top = d_elem_path.back();
381 if (top.child > 0) {
382 --top.child;
383 descend_elem_right(nth_child(top.elem, top.child));
384 return *this;
385 }
386 d_elem_path.pop_back();
387 }
388 // Step 2: ElemFrames exhausted — retreat across the spine structure.
389 retreat_spine();
390 return *this;
391 }
392
393 auto operator--(int) -> FingerTreeIterator {
394 auto copy = *this;
395 --*this;
396 return copy;
397 }
398
399 // -----------------------------------------------------------------------
400 // Comparison
401 // -----------------------------------------------------------------------
402
403 // Precondition: both iterators are from the same tree snapshot.
404 auto operator==(const FingerTreeIterator& other) const -> bool {
405 return d_absolute_index == other.d_absolute_index;
406 }

413template <typename T, typename MONOID, typename MP, typename Alloc>
414auto begin(const FingerTree<T, MONOID, MP, Alloc>& t) -> FingerTreeIterator<T, MONOID, MP, Alloc> {
415 // Compute tree size: O(1) for default measure, O(N) otherwise.
416 std::size_t n = 0;
417 if constexpr (FingerTree<T, MONOID, MP, Alloc>::has_unit_count_measure) {
418 n = t.measure();
419 } else {
420 t.for_each([&](const T&) { ++n; });
421 }
422 return FingerTreeIterator<T, MONOID, MP, Alloc>::make_begin(t, n);
423}
424
425template <typename T, typename MONOID, typename MP, typename Alloc>
426auto end(const FingerTree<T, MONOID, MP, Alloc>& t) -> FingerTreeIterator<T, MONOID, MP, Alloc> {
427 std::size_t n = 0;
428 if constexpr (FingerTree<T, MONOID, MP, Alloc>::has_unit_count_measure) {
429 n = t.measure();
430 } else {
431 t.for_each([&](const T&) { ++n; });
432 }
433 return FingerTreeIterator<T, MONOID, MP, Alloc>::make_end(t, n);
434}
