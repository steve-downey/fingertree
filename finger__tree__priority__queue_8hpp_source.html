61class FingerTreePriorityQueue {

66 public:

97 auto pop_min() const -> std::optional<std::pair<T, FingerTreePriorityQueue>> {
98 auto tag = d_tree.measure();
99 if (!tag.d_min.d_value.has_value())
100 return std::nullopt;
101 T global_min = *tag.d_min.d_value;
102 auto sp = d_tree.split([global_min](const PriorityTag<T>& p) {
103 return p.d_min.d_value.has_value() && *p.d_min.d_value <= global_min;
104 });
105 if (!sp.has_value())
106 return std::nullopt;
107 return std::pair<T, FingerTreePriorityQueue>{sp->d_pivot,
108 FingerTreePriorityQueue{Tree::concat(sp->d_left, sp->d_right)}};
109 }
110
111 auto pop_max() const -> std::optional<std::pair<T, FingerTreePriorityQueue>> {
112 auto tag = d_tree.measure();
113 if (!tag.d_max.d_value.has_value())
114 return std::nullopt;
115 T global_max = *tag.d_max.d_value;
116 auto sp = d_tree.split([global_max](const PriorityTag<T>& p) {
117 return p.d_max.d_value.has_value() && *p.d_max.d_value >= global_max;
118 });
119 if (!sp.has_value())
120 return std::nullopt;
121 return std::pair<T, FingerTreePriorityQueue>{sp->d_pivot,
122 FingerTreePriorityQueue{Tree::concat(sp->d_left, sp->d_right)}};
123 }
124
125 auto to_vector() const -> std::vector<T> { return d_tree.flatten(); }
126};

203 template <class APPLICATIVE, class F>
204 auto traverse(this auto&&,
205 const APPLICATIVE& applicative,
206 F&& function,
207 const beman::fingertree::tree::FingerTreePriorityQueue<T, Tree>& queue) {
208 using Context = remove_cvref_t<std::invoke_result_t<F, const T&>>;
209 using U = beman::fingertree::applicative_value_t<Context>;
210
211 auto accumulated = applicative.pure(std::vector<U>{});
212
213 for (const auto& value : queue.to_vector()) {
214 auto lifted = std::invoke(function, value);
215 accumulated = applicative.invoke(
216 [](std::vector<U> values, U element) {
217 values.push_back(std::move(element));
218 return values;
219 },
220 std::move(accumulated),
221 std::move(lifted));
222 }
223
224 return applicative.invoke(
225 [](std::vector<U> values) {
226 return beman::fingertree::tree::FingerTreePriorityQueue<U>::from_values(std::move(values));
227 },
228 std::move(accumulated));
229 }
230};
