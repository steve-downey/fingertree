27template <typename T>
28struct optional_min_monoid {
29 using value_type = std::optional<T>;
30
31 auto identity() const -> std::optional<T> { return std::nullopt; }
32
33 auto combine(const std::optional<T>& lhs, const std::optional<T>& rhs) const -> std::optional<T> {
34 if (!lhs.has_value())
35 return rhs;
36 if (!rhs.has_value())
37 return lhs;
38 return *lhs <= *rhs ? lhs : rhs;
39 }
40};

91template <typename T, typename Tree = FingerTree<T, PriorityMonoid<T>, PriorityMeasure<T>>>
92class FingerTreePriorityQueue {
93 Tree d_tree;

95 explicit FingerTreePriorityQueue(Tree tree) : d_tree(std::move(tree)) {}

97 public:
98 FingerTreePriorityQueue() : d_tree(Tree{}) {}
99
100 static auto from_values(std::vector<T> values) -> FingerTreePriorityQueue {
101 return FingerTreePriorityQueue{Tree::from_sequence(std::move(values))};
102 }
103
104 auto empty() const -> bool { return d_tree.is_empty(); }
105
107 auto size() const -> std::size_t { return std::get<0>(d_tree.measure()); }
108
109 auto min() const -> std::optional<T> { return std::get<1>(d_tree.measure()); }
110
111 auto max() const -> std::optional<T> { return std::get<2>(d_tree.measure()); }
112
113 auto push(T value) const -> FingerTreePriorityQueue {
114 return FingerTreePriorityQueue{d_tree.snoc(std::move(value))};
115 }
116
117 auto pop_min() const -> std::optional<std::pair<T, FingerTreePriorityQueue>> {
118 auto current = min();
119 if (!current.has_value())
120 return std::nullopt;
121 T global_min = *current;
122 auto sp = d_tree.split([global_min](const PriorityTag<T>& p) {
123 const auto& m = std::get<1>(p);
124 return m.has_value() && *m <= global_min;
125 });
126 if (!sp.has_value())
127 return std::nullopt;
128 return std::pair<T, FingerTreePriorityQueue>{sp->d_pivot,
129 FingerTreePriorityQueue{Tree::concat(sp->d_left, sp->d_right)}};
130 }
131
132 auto pop_max() const -> std::optional<std::pair<T, FingerTreePriorityQueue>> {
133 auto current = max();
134 if (!current.has_value())
135 return std::nullopt;
136 T global_max = *current;
137 auto sp = d_tree.split([global_max](const PriorityTag<T>& p) {
138 const auto& m = std::get<2>(p);
139 return m.has_value() && *m >= global_max;
140 });
141 if (!sp.has_value())
142 return std::nullopt;
143 return std::pair<T, FingerTreePriorityQueue>{sp->d_pivot,
144 FingerTreePriorityQueue{Tree::concat(sp->d_left, sp->d_right)}};
145 }
146
147 auto to_vector() const -> std::vector<T> { return d_tree.flatten(); }
148};

154template <class T, class Tree>
155struct FingerTreePriorityQueueFoldableImpl {
156 using element_type = T;
157
158 // The monoid is an instance object over the result type, passed by
159 // value; the Impl never chooses one (beman.transpose
160 // docs/decisions.md#monoid-selection).
161 template <class F, class RESULT_MONOID>
162 auto fold_map(this auto&&,
163 F&& function,
164 const beman::fingertree::tree::FingerTreePriorityQueue<T, Tree>& queue,
165 RESULT_MONOID monoid) -> remove_cvref_t<std::invoke_result_t<F&, const T&>> {
166 using Result = remove_cvref_t<std::invoke_result_t<F&, const T&>>;
167 return std::ranges::fold_left(
168 queue.to_vector(), Result(monoid.identity()), [&](Result acc, const auto& value) {
169 return monoid.combine(std::move(acc), std::invoke(function, value));
170 });
171 }
172};

181template <class T, class Tree>
182struct FingerTreePriorityQueueTraversableImpl {

185 template <class APPLICATIVE, class F>
186 auto traverse(this auto&&,
187 const APPLICATIVE& applicative,
188 F&& function,
189 const beman::fingertree::tree::FingerTreePriorityQueue<T, Tree>& queue) {
190 using Context = remove_cvref_t<std::invoke_result_t<F, const T&>>;
191 using U = beman::fingertree::applicative_value_t<Context>;
192
193 auto accumulated = applicative.pure(std::vector<U>{});
194
195 for (const auto& value : queue.to_vector()) {
196 auto lifted = std::invoke(function, value);
197 accumulated = applicative.invoke(
198 [](std::vector<U> values, U element) {
199 values.push_back(std::move(element));
200 return values;
201 },
202 std::move(accumulated),
203 std::move(lifted));
204 }
205
206 return applicative.invoke(
207 [](std::vector<U> values) {
208 return beman::fingertree::tree::FingerTreePriorityQueue<U>::from_values(std::move(values));
209 },
210 std::move(accumulated));
211 }
212};
