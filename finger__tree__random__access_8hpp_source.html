5// Persistent random-access sequence backed by a finger tree.
6//
7// The Tree template parameter accepts any fully-specialised finger tree type
8// whose measure is an element count (sum_monoid<std::size_t> over
9// UnitMeasure):
10//
11// FingerTreeRandomAccess<int> // FingerTree<int>-backed (default)
12// FingerTreeRandomAccess<int, SomeOtherTree<int>> // any tree with the same measure
13//
14// This mirrors std::stack<T, Container>: the element type T plus an optional
15// fully-specialised backing tree. Reason for the full-specialisation approach
16// (not template-template): different finger tree variants have different
17// template signatures; taking the complete instantiation is robust to drift.

46template <typename T,
47 typename Tree =
48 FingerTree<T, beman::fingertree::typeclass::sum_monoid<std::size_t>, UnitMeasure<T, std::size_t>>>
49class FingerTreeRandomAccess {
50 Tree d_tree;
51
52 public:
54 FingerTreeRandomAccess() : d_tree(Tree{}) {}
55
57 explicit FingerTreeRandomAccess(Tree tree) : d_tree(std::move(tree)) {}
58
60 static auto from_sequence(std::vector<T> values) -> FingerTreeRandomAccess {
61 return FingerTreeRandomAccess(Tree::from_sequence(std::move(values)));
62 }
63
65 auto size() const -> std::size_t { return d_tree.measure(); }
66
68 auto empty() const -> bool { return d_tree.is_empty(); }
69
72 auto at(std::size_t index) const -> std::optional<T> {
73 if (index >= size())
74 return std::nullopt;
75 auto sp = d_tree.split([index](std::size_t prefix) { return prefix > index; });
76 if (!sp.has_value())
77 return std::nullopt;
78 return sp->d_pivot;
79 }
80
83 auto push_back(T value) const -> FingerTreeRandomAccess {
84 return FingerTreeRandomAccess(d_tree.snoc(std::move(value)));
85 }
86
89 auto push_front(T value) const -> FingerTreeRandomAccess {
90 return FingerTreeRandomAccess(d_tree.cons(std::move(value)));
91 }
92
95 auto insert(std::size_t index, T value) const -> FingerTreeRandomAccess {
96 auto parts = d_tree.split_at([index](std::size_t prefix) { return prefix > index; });
97 return FingerTreeRandomAccess(
98 Tree::concat(Tree::concat(parts.d_left, Tree::leaf(std::move(value))), parts.d_right));
99 }
100
104 auto erase(std::size_t index) const -> FingerTreeRandomAccess {
105 if (index >= size())
106 return *this;
107 auto sp = d_tree.split([index](std::size_t prefix) { return prefix > index; });
108 if (!sp.has_value())
109 return *this;
110 return FingerTreeRandomAccess(Tree::concat(sp->d_left, sp->d_right));
111 }
112
116 auto update(std::size_t index, T value) const -> FingerTreeRandomAccess {
117 if (index >= size())
118 return *this;
119 auto sp = d_tree.split([index](std::size_t prefix) { return prefix > index; });
120 if (!sp.has_value())
121 return *this;
122 return FingerTreeRandomAccess(Tree::concat(sp->d_left.snoc(std::move(value)), sp->d_right));
123 }
124
126 auto to_vector() const -> std::vector<T> { return d_tree.flatten(); }
127};

134template <class T, class Tree>
135struct FingerTreeRandomAccessFoldableImpl {
136 using element_type = T;
137
138 // The monoid is an instance object over the result type, passed by
139 // value; the Impl never chooses one (beman.transpose
140 // docs/decisions.md#monoid-selection).
141 template <class F, class RESULT_MONOID>
142 auto fold_map(this auto&&,
143 F&& function,
144 const beman::fingertree::tree::FingerTreeRandomAccess<T, Tree>& sequence,
145 RESULT_MONOID monoid) -> remove_cvref_t<std::invoke_result_t<F&, const T&>> {
146 using Result = remove_cvref_t<std::invoke_result_t<F&, const T&>>;
147 return std::ranges::fold_left(
148 sequence.to_vector(), Result(monoid.identity()), [&](Result acc, const auto& value) {
149 return monoid.combine(std::move(acc), std::invoke(function, value));
150 });
151 }
152};

168template <class T, class Tree>
169struct FingerTreeRandomAccessTraversableImpl {
170 using element_type = T;
171
172 template <class APPLICATIVE, class F>
173 auto traverse(this auto&&,
174 const APPLICATIVE& applicative,
175 F&& function,
176 const beman::fingertree::tree::FingerTreeRandomAccess<T, Tree>& sequence) {
177 using Context = remove_cvref_t<std::invoke_result_t<F, const T&>>;
178 using U = beman::fingertree::applicative_value_t<Context>;
179
180 auto accumulated = applicative.pure(std::vector<U>{});
181
182 for (const auto& value : sequence.to_vector()) {
183 auto lifted = std::invoke(function, value);
184 accumulated = applicative.invoke(
185 [](std::vector<U> values, U element) {
186 values.push_back(std::move(element));
187 return values;
188 },
189 std::move(accumulated),
190 std::move(lifted));
191 }
192
193 return applicative.invoke(
194 [](std::vector<U> values) {
195 return beman::fingertree::tree::FingerTreeRandomAccess<U>::from_sequence(std::move(values));
196 },
197 std::move(accumulated));
198 }
199};
