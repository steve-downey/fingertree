5// Persistent random-access sequence backed by a finger tree.
6//
7// The Tree template parameter accepts any fully-specialised finger tree type
8// whose measure is an element count (size_t, UnitMeasure):
9//
10// FingerTreeRandomAccess<int> // -backed (default,
11// correct at all sizes) FingerTreeRandomAccess<int, FingerTree2<int>> //
12// FT2-backed (kMaxDepth hazard at N > ~2000)
13//
14// This mirrors std::stack<T, Container>: the element type T plus an optional
15// fully-specialised backing tree. Reason for the full-specialisation approach
16// (not template-template): different finger tree variants have different
17// template signatures; taking the complete instantiation is robust to drift.

46template <typename T, typename Tree = FingerTree<T, std::size_t, UnitMeasure<T, std::size_t>>>
47class FingerTreeRandomAccess {
48 Tree d_tree;
49
50 public:
52 FingerTreeRandomAccess() : d_tree(Tree{}) {}
53
55 explicit FingerTreeRandomAccess(Tree tree) : d_tree(std::move(tree)) {}
56
58 static auto from_sequence(std::vector<T> values) -> FingerTreeRandomAccess {
59 return FingerTreeRandomAccess(Tree::from_sequence(std::move(values)));
60 }
61
63 auto size() const -> std::size_t { return d_tree.measure(); }
64
66 auto empty() const -> bool { return d_tree.is_empty(); }
67
70 auto at(std::size_t index) const -> std::optional<T> {
71 if (index >= size())
72 return std::nullopt;
73 auto sp = d_tree.split([index](std::size_t prefix) { return prefix > index; });
74 if (!sp.has_value())
75 return std::nullopt;
76 return sp->d_pivot;
77 }
78
81 auto push_back(T value) const -> FingerTreeRandomAccess {
82 return FingerTreeRandomAccess(d_tree.snoc(std::move(value)));
83 }
84
87 auto push_front(T value) const -> FingerTreeRandomAccess {
88 return FingerTreeRandomAccess(d_tree.cons(std::move(value)));
89 }
90
93 auto insert(std::size_t index, T value) const -> FingerTreeRandomAccess {
94 auto parts = d_tree.split_at([index](std::size_t prefix) { return prefix > index; });
95 return FingerTreeRandomAccess(
96 Tree::concat(Tree::concat(parts.d_left, Tree::leaf(std::move(value))), parts.d_right));
97 }
98
102 auto erase(std::size_t index) const -> FingerTreeRandomAccess {
103 if (index >= size())
104 return *this;
105 auto sp = d_tree.split([index](std::size_t prefix) { return prefix > index; });
106 if (!sp.has_value())
107 return *this;
108 return FingerTreeRandomAccess(Tree::concat(sp->d_left, sp->d_right));
109 }
110
114 auto update(std::size_t index, T value) const -> FingerTreeRandomAccess {
115 if (index >= size())
116 return *this;
117 auto sp = d_tree.split([index](std::size_t prefix) { return prefix > index; });
118 if (!sp.has_value())
119 return *this;
120 return FingerTreeRandomAccess(Tree::concat(sp->d_left.snoc(std::move(value)), sp->d_right));
121 }
122
124 auto to_vector() const -> std::vector<T> { return d_tree.flatten(); }
125};

132template <class T, class Tree>
133struct FingerTreeRandomAccessFoldableImpl {
134 template <class F>
135 auto fold_map(this auto&&, F&& function, const beman::fingertree::tree::FingerTreeRandomAccess<T, Tree>& sequence)
136 -> remove_cvref_t<std::invoke_result_t<F, const T&>> {
137 using Result = remove_cvref_t<std::invoke_result_t<F, const T&>>;
138 return std::ranges::fold_left(sequence.to_vector(),
139 beman::fingertree::typeclass::monoid_v<Result>.identity(),
140 [&](Result acc, const auto& value) {
141 return beman::fingertree::typeclass::monoid_v<Result>.combine(
142 std::move(acc), std::invoke(function, value));
143 });
144 }
145};

163template <class T, class Tree>
164struct FingerTreeRandomAccessTraversableImpl {
165 using element_type = T;
166
167 template <class APPLICATIVE, class F>
168 auto traverse(this auto&&,
169 const APPLICATIVE& applicative,
170 F&& function,
171 const beman::fingertree::tree::FingerTreeRandomAccess<T, Tree>& sequence) {
172 using Context = remove_cvref_t<std::invoke_result_t<F, const T&>>;
173 using U = beman::fingertree::applicative_value_t<Context>;
174
175 auto accumulated = applicative.pure(std::vector<U>{});
176
177 for (const auto& value : sequence.to_vector()) {
178 auto lifted = std::invoke(function, value);
179 accumulated = applicative.invoke(
180 [](std::vector<U> values, U element) {
181 values.push_back(std::move(element));
182 return values;
183 },
184 std::move(accumulated),
185 std::move(lifted));
186 }
187
188 return applicative.invoke(
189 [](std::vector<U> values) {
190 return beman::fingertree::tree::FingerTreeRandomAccess<U>::from_sequence(std::move(values));
191 },
192 std::move(accumulated));
193 }
194};

199};
