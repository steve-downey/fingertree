17// Identity function for fold composition - no type erasure needed
18template <class STATE>
19struct IdentityFoldFunc {
20 constexpr auto operator()(STATE s) const -> STATE { return s; }

23// Compose two fold functions without type erasure
24template <class F1, class F2>
25struct ComposedFoldFunc {
26 F1 d_f1;
27 F2 d_f2;
28
29 template <class STATE>
30 auto operator()(STATE s) const -> STATE {
31 return d_f2(d_f1(std::move(s)));
32 }
33};

191// Foldable pattern invariants:
192// - Generic entry point is fold_map(F, T) via foldable_typeclass<T>.
193// - Instances provide an object with fold_map(F, T) and specialize
194// foldable_typeclass<Concrete>.
195// - Derived APIs (length, fold_left, fold_right, combine_all, any_of, all_of,
196// empty, to_vector, find_first) live on the same looked-up object.
197// - Traversal order is instance-defined but must be coherent per instance.
198
204template <class Impl>
205struct Foldable : protected Impl {
206 static_assert(!std::is_same_v<Impl, std::false_type>,
207 "No foldable_typeclass<T> specialization found. "
208 "Specialize beman::fingertree::foldable_typeclass<T> for your type T "
209 "and provide fold_map(F, T) or fold_right(T, STATE, F) + "
210 "element_type.");
211 // Alternate-core: Impl provides either fold_map or fold_right as primitive.
212 // The Map class's using-declaration selects which; the base derives the
213 // other. Haskell equivalent: {-# MINIMAL foldMap | foldr #-}
214
215 // Derived fold_map from fold_right. Active when a fold_right-primitive
216 // Impl's using-declaration shadows the base's derived fold_right with the
217 // real one. Requires element_type to deduce the monoid result type. foldMap
218 // f = foldr (\x acc -> f x <> acc) mempty
219 template <class F, class T>
220 auto fold_map(this auto&& self, F&& function, T&& value)
221 requires requires { typename Impl::element_type; }
222 {
223 using Result = remove_cvref_t<std::invoke_result_t<F, const typename Impl::element_type&>>;
224 return self.fold_right(
225 std::forward<T>(value), monoid_identity<Result>(), [&function](const auto& elem, Result acc) {
226 return monoid_combine(std::invoke(function, elem), std::move(acc));
227 });
228 }

244 template <class T, class STATE, class F>
245 auto fold_left(this auto&& self, T&& value, STATE initial_state, F&& function) {
246 using StateType = remove_cvref_t<STATE>;
247 auto step = std::forward<F>(function);
248
249 const auto program = self.fold_map(
250 [&step](const auto& x) {
251 using ValueType = remove_cvref_t<decltype(x)>;
252 return detail::LeftFoldProgram<StateType>{
253 [x_copy = ValueType(x), &step](StateType s) { return std::invoke(step, std::move(s), x_copy); }};
254 },
255 std::forward<T>(value));
256
257 return program(StateType(std::move(initial_state)));
258 }

263 template <class T, class STATE, class F>
264 auto fold_right(this auto&& self, T&& value, STATE initial_state, F&& function) {
265 using StateType = remove_cvref_t<STATE>;
266 auto step = std::forward<F>(function);
267
268 const auto program = self.fold_map(

271 return detail::RightFoldProgram<StateType>{

273 },
274 std::forward<T>(value));
