20// This header is a copy of beman.transpose's monoid.hpp with the namespace
21// renamed; keep the two in step rather than editing one.
22//
23// Monoid pattern invariants:
24// - A monoid is an *instance object*: anything with `identity()` and
25// `combine(lhs, rhs)` over one value type. `monoid_object<M, T>` is the
26// concept.
27// - Monoid<T> is the registry: the one canonical instance for a type, where
28// one exists (string and vector concatenation). monoid_v<T> is the lookup
29// object generic algorithms default to when the caller passes nothing.
30// - Where a type carries more than one monoid -- numbers, booleans -- no
31// instance is registered; the caller passes one (`sum_monoid<int>{}`,
32// `max_monoid<int>{}`, `any_monoid{}`, ...). The value type is never
33// wrapped to select a monoid. See beman.transpose docs/decisions.md#monoid-selection.
34// - Instances may carry state. The library may copy them and pass them by
35// value; the expectation (not a requirement) is that they are shaped like
36// stateless objects whose calls resolve statically.
37// - identity and combine must stay coherent for a single associative law
38// domain.
39

114template <class T>
115struct max_monoid {
116 using value_type = T;
117
118 constexpr auto identity() const -> T {
119 if constexpr (std::numeric_limits<T>::has_infinity) {
120 return -std::numeric_limits<T>::infinity();
121 } else {
122 return std::numeric_limits<T>::lowest();
123 }
124 }
125
126 constexpr auto combine(const T& lhs, const T& rhs) const -> T { return lhs < rhs ? rhs : lhs; }

194template <class M>
195struct dual_monoid {
196 using value_type = monoid_value_t<M>;
197
198 M d_inner{};
199
200 constexpr auto identity() const -> value_type { return d_inner.identity(); }
201
202 constexpr auto combine(const value_type& lhs, const value_type& rhs) const -> value_type {
203 return d_inner.combine(rhs, lhs);
204 }
205};

210template <class... M>
211struct tuple_monoid {
212 using value_type = std::tuple<monoid_value_t<M>...>;
213
214 std::tuple<M...> d_inners{};
215
216 constexpr auto identity() const -> value_type {
217 return std::apply([](const M&... inner) { return value_type{inner.identity()...}; }, d_inners);
218 }
219
220 constexpr auto combine(const value_type& lhs, const value_type& rhs) const -> value_type {
221 return [&]<std::size_t... I>(std::index_sequence<I...>) {
222 return value_type{std::get<I>(d_inners).combine(std::get<I>(lhs), std::get<I>(rhs))...};
223 }(std::index_sequence_for<M...>{});
224 }
225};
