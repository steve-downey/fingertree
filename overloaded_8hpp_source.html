5// overloaded<Ts...> — aggregate visitor for std::visit.
6//
7// Usage:
8// std::visit(overloaded{
9// [](int x) { ... },
10// [](std::string s) { ... },
11// }, v);
12//
13// The consteval catch-all fires a static_assert at compile time if std::visit
14// encounters an alternative not covered by the explicit cases. This turns
15// variant exhaustiveness into a hard compile error rather than a silent
16// default/no-op. Adding a new alternative to a variant without handling it
17// everywhere is caught immediately.
18//
19// No explicit deduction guide is needed: C++20 CTAD for aggregates deduces
20// overloaded<F1, F2, ...> from the constructor arguments.
21
22namespace beman::fingertree::fixpoint {
23
24template <typename... Ts>
25struct overloaded : Ts... {
26 using Ts::operator()...;

28 consteval void operator()(auto) const {
29 static_assert(false, "overloaded: unhandled variant alternative — add a case");
30 }
31};
